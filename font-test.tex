% Helper: print base+each mark from a list
\cs_new_protected:Nn \__dm_print_series:Nn
  {
    \clist_map_inline:Nn #1 { #2##1\space }
  }

% Public macro:
% One line per base:
%   base | top-mark series | bottom-mark series | complex series
\ExplSyntaxOn

% Print base+mark with fixed spacing
\cs_new_protected:Nn \__dm_print_single_series_spaced:Nn
  {
    \clist_map_inline:Nn #1 { #2##1 \GlyphSep }
  }

% One line per base:
%   base | TOP marks | BOTTOM marks | COMPLEX marks
\newcommand{\SingleMarksTest}{%
  \par\noindent
  \clist_map_inline:Nn \g__dm_bases_clist
    {
      % Column 0: plain base
      ##1

      % Column 1: top diacritics
      \tab
      \__dm_print_single_series_spaced:Nn \g__dm_topmarks_clist {##1}

      % Column 2: bottom diacritics
      \tab
      \__dm_print_single_series_spaced:Nn \g__dm_botmarks_clist {##1}

      % Column 3: complex diacritics
      \tab
      \__dm_print_single_series_spaced:Nn \g__dm_complex_clist {##1}

      \par
    }%
}

% Print base+mark+extra with fixed separator
\cs_new_protected:Nn \__dm_print_combined_series_spaced:Nnn
  { \clist_map_inline:Nn #1 { #2##1#3 \GlyphSep } }

% One line per base:
%   base | TOP ( +acute, +dotbelow ) | BOTTOM ( +acute, +dotbelow ) | COMPLEX (plain)
\newcommand{\CombinedMarksTest}{%
  \par\noindent
  \clist_map_inline:Nn \g__dm_bases_clist
    {
      % Column 0: the plain base at line start
      ##1

      % Column 1: TOP diacritics combined with extra acute ABOVE, then with dot BELOW
      \tab
      \__dm_print_combined_series_spaced:Nnn \g__dm_topmarks_clist {##1} { \__dm_acute_above: }
      \tab
      \__dm_print_combined_series_spaced:Nnn \g__dm_topmarks_clist {##1} { \__dm_dot_below: }

      % Column 2: BOTTOM diacritics combined with extra acute ABOVE, then with dot BELOW
      \tab
      \__dm_print_combined_series_spaced:Nnn \g__dm_botmarks_clist {##1} { \__dm_acute_above: }
      \tab
      \__dm_print_combined_series_spaced:Nnn \g__dm_botmarks_clist {##1} { \__dm_dot_below: }

%       % Column 3: COMPLEX marks (not combined)

%       \__dm_print_series_spaced:Nn \g__dm_complex_clist {##1}

      \par
    }%
}
